# provider_manifests — what cmake/EngineProviders.cmake refuses, and why (WO-085).
# Configured once per CASE by ctest (tests/CMakeLists.txt); nothing is built.
# Each case writes its manifests here and declares them; the test checks the
# configure output for the message that names the problem.
cmake_minimum_required(VERSION 3.20)
project(provider_manifest_cases LANGUAGES CXX)
include(${ENGINE_ROOT}/cmake/EngineProviders.cmake)

set(D ${CMAKE_BINARY_DIR}/m)
file(MAKE_DIRECTORY ${D})
function(manifest kind name platforms extra)
    file(WRITE ${D}/${kind}-${name}.provider.json
        "{ \"kind\": \"${kind}\", \"name\": \"${name}\", \"contract\": \"engine-plugin\",
           \"platforms\": ${platforms} ${extra} }")
endfunction()
set(ALL "[\"windows\", \"linux\", \"macos\", \"android\", \"ios\", \"web\"]")
file(WRITE ${CMAKE_BINARY_DIR}/t.cpp "")
add_library(target_under_test STATIC ${CMAKE_BINARY_DIR}/t.cpp)

if(CASE STREQUAL "ok")
    manifest(physics a "${ALL}" "")
    engine_provider(MANIFEST ${D}/physics-a.provider.json)
elseif(CASE STREQUAL "unknown_platform")
    manifest(physics a "[\"playstation\"]" "")
    engine_provider(MANIFEST ${D}/physics-a.provider.json)
elseif(CASE STREQUAL "no_platforms")
    manifest(physics a "[]" "")
    engine_provider(MANIFEST ${D}/physics-a.provider.json)
elseif(CASE STREQUAL "unknown_fact")
    manifest(physics a "${ALL}" ", \"needs\": [\"quantum\"]")
    engine_provider(MANIFEST ${D}/physics-a.provider.json)
elseif(CASE STREQUAL "unknown_kind")
    manifest(teleport a "${ALL}" "")
    engine_provider(MANIFEST ${D}/teleport-a.provider.json)
elseif(CASE STREQUAL "wrong_file_name")
    manifest(physics a "${ALL}" "")
    file(RENAME ${D}/physics-a.provider.json ${D}/physics-b.provider.json)
    engine_provider(MANIFEST ${D}/physics-b.provider.json)
elseif(CASE STREQUAL "unknown_contract")
    file(WRITE ${D}/physics-a.provider.json
        "{ \"kind\": \"physics\", \"name\": \"a\", \"contract\": \"nope\", \"platforms\": ${ALL} }")
    engine_provider(MANIFEST ${D}/physics-a.provider.json)
elseif(CASE STREQUAL "declared_twice")
    manifest(physics a "${ALL}" "")
    engine_provider(MANIFEST ${D}/physics-a.provider.json)
    engine_provider(MANIFEST ${D}/physics-a.provider.json)
elseif(CASE STREQUAL "provider_links_provider")
    manifest(physics a "${ALL}" "")
    manifest(audio b "${ALL}" "")
    engine_provider(MANIFEST ${D}/audio-b.provider.json LINKS engine_provider_physics_a)
    engine_provider(MANIFEST ${D}/physics-a.provider.json)
elseif(CASE STREQUAL "other_platform")
    manifest(physics a "[\"web\"]" "")
    if(ENGINE_PLATFORM STREQUAL "web")
        manifest(physics a "[\"windows\"]" "")
    endif()
    engine_provider(MANIFEST ${D}/physics-a.provider.json)
elseif(CASE STREQUAL "missing_sdk")
    manifest(physics a "${ALL}" ", \"sdk\": [\"FAKE_PHYSICS_SDK\"]")
    engine_provider(MANIFEST ${D}/physics-a.provider.json)
elseif(CASE STREQUAL "server_flavour")
    manifest(renderer a "${ALL}" ", \"flavours\": [\"client\"]")
    file(WRITE ${D}/renderer-a.provider.json
        "{ \"kind\": \"renderer\", \"name\": \"a\", \"contract\": \"renderer\",
           \"platforms\": ${ALL}, \"flavours\": [\"client\"] }")
    engine_provider(MANIFEST ${D}/renderer-a.provider.json)
    engine_providers_apply(target_under_test FLAVOUR server)
    return()
else()
    message(FATAL_ERROR "unknown CASE '${CASE}'")
endif()
engine_providers_apply(target_under_test FLAVOUR client)
